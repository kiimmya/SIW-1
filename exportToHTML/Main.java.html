Main.java 
import java.util.*;

public class Main {

    public static void main(String[] args) {

        System.out.println(" RECIPE COLLECTION ORGANIZER 3.0");

        MainCourses chickenCurry = new MainCourses(1, "Chicken Curry", 45, 4, "Chicken", false);
        MainCourses beefSteak = new MainCourses(2, "Beef Steak", 30, 2, "Beef", false);
        Desserts chocolateCake = new Desserts(3, "Chocolate Cake", 60, 8, "Chocolate", true);
        Desserts fruitSalad = new Desserts(4, "Fruit Salad", 15, 3, "Fruit", true);
        Pastry applePie = new Pastry(5, "Apple Pie", 50, 6, "Wheat", true);

        RecipeCollection collection = new RecipeCollection();

        System.out.println("\n--- ADD RECIPES ---");

        collection.addRecipe(chickenCurry);
        collection.addRecipe(beefSteak);
        collection.addRecipe(chocolateCake);
        collection.addRecipe(fruitSalad);
        collection.addRecipe(applePie);


// Create users
        User user = new User("Diana", UserRole.USER);
        User admin = new User("Admin", UserRole.ADMIN);


// Admin processes requests
        System.out.println("\n--- User tries to process request ---");
        collection.processNextRequest(user);

        System.out.println("\n--- Admin processes requests ---");
        collection.processNextRequest(admin);
        collection.processNextRequest(admin);


        System.out.println("\n--- ALL RECIPES ---");
        collection.showRecipes();

        System.out.println("\n--- SPECIAL RECIPE DETAILS ---");
        beefSteak.grill();
        chocolateCake.decorate();
        applePie.bake();


// SEARCH BY ID
        System.out.println("\n--- SEARCH BY ID ---");
        Recipe found = collection.findRecipeById(3);

        if (found != null) {
            found.printInfo();
        } else {
            System.out.println("Recipe not found.");
        }


// SEARCH BY NAME
        System.out.println("\n--- SEARCH BY NAME ---");
        List<Recipe> searchResults = collection.searchByName("cake");

        for (Recipe recipe : searchResults) {
            System.out.println(recipe.getName());
        }


// FILTER BY CATEGORY
        System.out.println("\n--- FILTER BY CATEGORY ---");
        List<Recipe> desserts =
                collection.filterByCategory("Dessert");

        for (Recipe recipe : desserts) {
            System.out.println(recipe.getName());
        }


// COMPARABLE: SORT BY ID
        System.out.println("\n--- SORT BY ID ---");
        collection.sortById();


// SORT BY NAME
        System.out.println("\n--- SORT BY NAME ---");
        collection.sortRecipes(new RecipeNameComparator());


// SORT BY COOKING TIME
        System.out.println("\n--- SORT BY COOKING TIME ---");
        collection.sortRecipes(new CookingTimeComparator());


// SORT BY PORTIONS
        System.out.println("\n--- SORT BY PORTIONS ---");
        collection.sortRecipes(new PortionsComparator());


// SET: UNIQUE INGREDIENTS
        System.out.println("\n--- UNIQUE INGREDIENTS ---");
        collection.showIngredients();

// UPDATE RECIPE
        System.out.println("\n--- UPDATE RECIPE ---");
        Desserts updatedCake = new Desserts(3, "Chocolate Cake Deluxe", 70, 10, "Chocolate", true);
        collection.updateRecipe(3, updatedCake);
        Recipe updated = collection.findRecipeById(3);

        if (updated != null) {
            updated.printInfo();
        }

// QUEUE: USER SUBMITS REQUESTS
        System.out.println("\n--- REQUEST QUEUE ---");
        Desserts cheesecake = new Desserts(6, "Cheesecake", 55, 6, "Cream Cheese", true);
        RecipeRequest addRequest = new RecipeRequest(RecipeRequest.RequestType.ADD, 6, cheesecake, user);
        RecipeRequest deleteRequest = new RecipeRequest(RecipeRequest.RequestType.DELETE, 2, null, user);

        collection.submitRequest(addRequest);
        collection.submitRequest(deleteRequest);

        System.out.println("Pending requests: " + collection.getPendingRequestsCount());


// ANALYTICS
        System.out.println("\n--- ANALYTICS ---");
        collection.printRecipeCount();


// COMPOSITION
        System.out.println("\n--- RECIPE BOOK ---");
        collection.showRecipeBook();


// POLYMORPHISM
        System.out.println("\n--- POLYMORPHISM ---");
        for (Recipe recipe : collection.getRecipes()) {
            recipe.serve();
            recipe.adjustPortion();

            System.out.println();
        }


// DELETE RECIPE
        System.out.println("\n--- DELETE RECIPE ---");
        collection.deleteRecipe(4);


        System.out.println("\n--- FINAL RECIPES ---");
        collection.showRecipes();
        System.out.println("\nPROGRAM FINISHED.");
    }
}
